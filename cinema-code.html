Cinema ticket generator

A Tkinter booking app: pick a film and time, enter ticket numbers, add extras, and get the total with a group discount.

tkinter_cinema_app.py
import tkinter as tk
from tkinter import messagebox as mb

w = tk.Tk()
w.title("Cinema Booking")
w.geometry("370x1030")
movie = tk.StringVar(value="None")
snack1 = tk.IntVar(value=0)
glasses1 = tk.IntVar(value=0)
time = tk.StringVar(value="14:00")

# Frames
w1 = tk.LabelFrame(
    w,
    text="Cinema Booking",
    font=("Arial", 20, "bold"),
    fg="blue"
)
w1.grid(row=0, column=0, padx=10, pady=10)

w2 = tk.Frame(w)
w2.grid(row=1, column=0, padx=10, pady=10)

#name

tk.Label(w1, text="Name").grid(
    row=0, column=0, padx=5, pady=5
)

NameEntry = tk.Entry(w1, width=30)
NameEntry.grid(
    row=0, column=1, padx=5, pady=5
)

#movie

film = tk.Label(
    w1,
    text="Pick A Movie:",
    font=("Arial", 15, "bold"),
    fg="purple"
)
film.grid(row=1, column=0, columnspan=2, padx=5, pady=5)

movie1 = tk.Radiobutton(
    w1,
    text="Minecraft Movie",
    variable=movie,
    value="Minecraft Movie"
)
movie1.grid(row=2, column=0, columnspan=2)

movie2 = tk.Radiobutton(
    w1,
    text="Lord Verity",
    variable=movie,
    value="Lord Verity"
)
movie2.grid(row=3, column=0, columnspan=2)

movie3 = tk.Radiobutton(
    w1,
    text="Ryan Cooked",
    variable=movie,
    value="Ryan Cooked"
)
movie3.grid(row=4, column=0, columnspan=2)

#tickets

tk.Label(
    w1,
    text="Number Of Adults Tickets"
).grid(row=5, column=0, padx=5, pady=5)

NumAEntry = tk.Entry(w1, width=30)
NumAEntry.grid(row=5, column=1, padx=5, pady=5)

tk.Label(
    w1,
    text="Number Of Child Tickets"
).grid(row=6, column=0, padx=5, pady=5)

NumCEntry = tk.Entry(w1, width=30)
NumCEntry.grid(row=6, column=1, padx=5, pady=5)

#prices

tk.Label(
    w1,
    text="Adult Ticket Price: £10"
).grid(row=7, column=0, padx=5, pady=5)

    w1,
    text="Child Ticket Price: £6"
).grid(row=7, column=1, padx=5, pady=5)

#extras

snack = tk.Checkbutton(
    w1,
    text="Snack Pack +£4",

)

glasses = tk.Checkbutton(
    w1,

)

#time

tk.Label(
    w1,

tk.Label(
    w2,
    text="Total Price",
    font=("Arial", 12, "bold")
).grid(row=0, column=0, padx=5, pady=5)

result = tk.Label(
    w2,

    font=("Arial", 12, "bold")
)
result.grid(row=0, column=1, padx=5, pady=5)

stuff = tk.Label(w2, text="")
stuff.grid(row=1, column=0, columnspan=2, pady=5)

stuff2 = tk.Label(w2, text="")
stuff2.grid(row=2, column=0, columnspan=2, pady=5)

stuff3 = tk.Label(w2, text="")
stuff3.grid(row=3, column=0, columnspan=2, pady=5)

stuff4 = tk.Label(w2, text="")
stuff4.grid(row=4, column=0, columnspan=2, pady=5)

#validation

def validate_movie():
    if movie.get() == "None":
        stuff3.config(
            text="Error, please select a movie.",
            fg="red"
        )
        return False
    else:
        stuff3.config(
            text=f"Success, you selected: {movie.get()}",
            fg="green"
        )
        return True


#calc

def calc():
    validate_movie()
    name = NameEntry.get()
    if name == "":
        stuff.config(
            text="Please enter your name",
            fg="red"
        )
    else:
        stuff.config(
            text="Hello " + name,
            fg="green"
        )
    try:
        num1 = int(NumAEntry.get())
        num2 = int(NumCEntry.get())
    except ValueError:
        stuff2.config(
            text="Please enter valid whole numbers for tickets",
            fg="red"
        )
        return
    if not (1 <= num1 <= 10 and 1 <= num2 <= 10):
        stuff2.config(
            text="Please enter a whole number of tickets between 1 and 10",
            fg="red"
        )
        return
    thing = (num1 * 10) + (num2 * 6)
    if num1 + num2 >= 5:
        thing *= 0.9
        stuff4.config(
            text="10% discount applied due to 5+ tickets",
            fg="blue"
        )
    else:
        stuff4.config(text="")
    if snack1.get() == 1:
        thing += 4
    if glasses1.get() == 1:
        thing += 2
    result.config(
        text=f"£{thing:.2f}"
    )
    stuff2.config(
        text=f"Thank you for booking {num1 + num2} tickets with us",
        fg="green"
    )
    
calc_button = tk.Button(
    w2,
    text="Calculate",
    command=calc,
    bg="blue",
    fg="white"
)
calc_button.grid(
    row=5,
    column=0,
    columnspan=2,
    pady=10
)


#clear

def clear():
    stuff.config(text="")
    stuff2.config(text="")
    stuff3.config(text="")
    stuff4.config(text="")
    NameEntry.delete(0, tk.END)
    NumAEntry.delete(0, tk.END)
    NumCEntry.delete(0, tk.END)
    result.config(text="£0.00")
    movie.set("None")
    time.set("14:00")
    snack1.set(0)
    glasses1.set(0)

clear_button = tk.Button(
    w2,
    text="Clear",
    command=clear,
    bg="red"
)
clear_button.grid(row=6, column=0, pady=5)


#submit

def submit():
    choice = movie.get()
    name = NameEntry.get()
    try:
        num1 = int(NumAEntry.get())
        num2 = int(NumCEntry.get())
    except ValueError:
        mb.showerror(
            "Unconfirmed",
            "Please enter valid ticket numbers."
        )
        return
    if choice == "None":
        mb.showerror(
            "Unconfirmed",
            "Please select a movie."
        )
        return
    if name == "":
        mb.showerror(
            "Unconfirmed",
            "Please enter your name."
        )
        return
    if not (1 <= num1 <= 10 and 1 <= num2 <= 10):
        mb.showerror(
            "Unconfirmed",
            "Please enter between 1 and 10 tickets of each type."
        )
        return
    thing = (num1 * 10) + (num2 * 6)
    if num1 + num2 >= 5:
        thing *= 0.9
    if snack1.get() == 1:
        thing += 4
    if glasses1.get() == 1:
        thing += 2
    mb.showinfo(
        "Booking Confirmed",
        "Name: " + name + "\n"
        "Movie: " + choice + "\n"
        "Time: " + time.get() + "\n"
        "Adult tickets: " + str(num1) + "\n"
        "Child tickets: " + str(num2) + "\n"
        "Total tickets: " + str(num1 + num2) + "\n"
        "Total price: " + f"£{thing:.2f}"
    )

submit_button = tk.Button(
    w2,
    text="Submit",
    bg="green",
    command=submit
)
submit_button.grid(row=6, column=1, pady=5)


#logo

try:
    logo = tk.PhotoImage(file="popcoin.png")

except tk.TclError:

w.mainloop()
