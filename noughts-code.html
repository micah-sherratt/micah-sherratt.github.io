Tkinter noughts and crosses

A two-player noughts and crosses game built with Tkinter. Players take turns on a 3x3 grid, and the game announces a win or a draw.

tkinter_noughts_and_crosses.py
import tkinter as tk

w = tk.Tk()
w.title ("Noughts And Crosses")
w.geometry("325x280")
w1 = tk.LabelFrame(w, text="Noughts And Crosses", font=("Arial", 20, "bold"), fg="blue")
w1.grid(row=1, column=0, padx=10, pady=10)
turn = True
count = 0

def make_move(square):
    global turn
    global count
    if square["text"] != " ":
        return
    if turn:
        square.config(text="O")
        turn = False
    else:
        square.config(text="X")
        turn = True
    win()
    count = count + 1
    word = thing.cget("text")
    if count == 9 and word == "keep playing" :
        thing.config(text="draw")
        

square1 = tk.Button(w1, text=" ", bg="black", fg="white", width=5, height=2, command=lambda: make_move(square1))
